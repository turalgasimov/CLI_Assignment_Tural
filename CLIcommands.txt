code . -> to open the folder in vscode

mkdir <name> -> to create the folder {
	$ mkdir CLI_Assignment_Tural
}

cd  -> move to the folder {
	$ cd CLI_Assignment_Tural
}

clear -> clear the shell

touch -> create the files {
	$ touch notes.txt
}

history -> to check the command history 

echo -> to write something to the file {
	$ echo $'What time is it?\nIt\'s CLI time!\nWhat a rhyme!'
}

cp -> to copy the file {
	$ cp notes.txt docs/notes_copy.txt
}

mv -> to rename the file {
	$ mv docs/notes_copy.txt docs/study.txt
}

ls -> to list the directory {
	$ ls -la
	-a -> show all, including hidden
	-l -> show details (time created, size, permissions and etc.)
}

cat -> to get the content of the file
grep -> to search for the word {
	$ cat notes.txt | grep "CLI" > search_results.txt
}

ifconfig -> to get the network information

systeminfo/uname -a -> to get the system information

ping -> to test the connectivity
head -> to get the first lines
tail -> to the the last lines {
	$ ping -c 4 example.com | head -5 | tail -4 >> network.txt
}

chmod -> to change permissions {
	$ chmod 640 secute.txt

	- 6 (110) -> rw- (read and write) for user
	- 4 (100) -> r-- (read) for group
	- 0 (000) -> --- (none) for others
}

rm -> to remove a file {
	$ rm ghost.txt 2> errors.txt

	- while > directs the output of the command,
	- 2> directs the related error if command fails.

	- in UNIX environment, we have:
	- STDIN for input coded as 0,
	- STDOUT for output coded as 1,
	- STDERR for errors coded as 2;

	- error appears because we try to delete
	- a file that does not exist.
}
